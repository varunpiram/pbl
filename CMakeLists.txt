cmake_minimum_required(VERSION 3.20)
project(productivity_blocker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(blocker src/core.cpp src/macos.cpp)
target_include_directories(blocker PUBLIC include)
target_compile_options(blocker PRIVATE -Wall -Wextra -Wpedantic)
add_executable(blockerctl src/main.cpp)
target_link_libraries(blockerctl PRIVATE blocker)
enable_testing()
add_executable(blocker_tests tests/tests.cpp)
target_link_libraries(blocker_tests PRIVATE blocker)
add_test(NAME backend COMMAND blocker_tests)
